Keycloak으로 로그인

API 토큰 검증 테스트

# Keycloak에서 access token 발급 (Direct Access Grant)
curl -s -X POST 'http://localhost:8180/realms/demo/protocol/openid-connect/token' \
  -d 'grant_type=password' \
  -d 'client_id=security-sso-app' \
  -d 'client_secret=sso-app-secret' \
  -d 'username=demo' \
  -d 'password=demo123'

# API 호출
curl -H "Authorization: Bearer <access_token>" http://localhost:8091/api/me
